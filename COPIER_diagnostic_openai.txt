<?php
/**
 * DIAGNOSTIC OPENAI - À UPLOADER EN TANT QUE diagnostic_openai.php
 * Ce script teste directement l'API OpenAI pour identifier le problème
 */

// 1. Charger la configuration
require_once __DIR__ . '/config/config.php';
require_once __DIR__ . '/api/pre_analysis.php';
require_once __DIR__ . '/api/legal_facts_manager.php';

header('Content-Type: text/html; charset=utf-8');

echo "<h1>🔍 Diagnostic OpenAI API</h1>";
echo "<style>body{font-family:monospace;padding:20px;} .ok{color:green;} .error{color:red;} .warning{color:orange;} pre{background:#f4f4f4;padding:10px;border-radius:5px;}</style>";

// TEST 1 : Vérifier la clé API
echo "<h2>TEST 1 : Clé API OpenAI</h2>";
if (defined('OPENAI_API_KEY') && OPENAI_API_KEY !== 'sk-votre-cle-api-openai') {
    $key_preview = substr(OPENAI_API_KEY, 0, 7) . '...' . substr(OPENAI_API_KEY, -4);
    echo "<p class='ok'>✅ Clé API définie : {$key_preview}</p>";
    echo "<p>Longueur : " . strlen(OPENAI_API_KEY) . " caractères</p>";
} else {
    echo "<p class='error'>❌ Clé API non configurée ou invalide</p>";
    echo "<p><strong>ACTION REQUISE :</strong> Modifier config/config.php avec votre vraie clé API</p>";
    exit;
}

// TEST 2 : Test basique OpenAI (sans function calling)
echo "<h2>TEST 2 : Appel OpenAI Simple (sans tools)</h2>";
echo "<p>Question test : 'Bonjour'</p>";

$data = [
    'model' => OPENAI_MODEL,
    'messages' => [
        [
            'role' => 'system',
            'content' => 'Tu es un assistant. Réponds brièvement.'
        ],
        [
            'role' => 'user',
            'content' => 'Bonjour'
        ]
    ],
    'max_tokens' => 100,
    'temperature' => 0.3
];

$ch = curl_init('https://api.openai.com/v1/chat/completions');
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
curl_setopt($ch, CURLOPT_POST, true);
curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($data));
curl_setopt($ch, CURLOPT_HTTPHEADER, [
    'Content-Type: application/json',
    'Authorization: Bearer ' . OPENAI_API_KEY
]);

$response = curl_exec($ch);
$http_code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
curl_close($ch);

echo "<p>Code HTTP : <strong>{$http_code}</strong></p>";

if ($http_code === 200) {
    echo "<p class='ok'>✅ Appel OpenAI réussi</p>";
    $response_data = json_decode($response, true);
    echo "<p><strong>Réponse :</strong> " . ($response_data['choices'][0]['message']['content'] ?? 'N/A') . "</p>";
    echo "<details><summary>Réponse complète JSON</summary><pre>" . json_encode($response_data, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE) . "</pre></details>";
} else {
    echo "<p class='error'>❌ Erreur OpenAI</p>";
    echo "<pre>" . htmlspecialchars($response) . "</pre>";

    $error_data = json_decode($response, true);
    if (isset($error_data['error']['message'])) {
        echo "<p class='error'><strong>Message d'erreur :</strong> {$error_data['error']['message']}</p>";
    }

    if ($http_code === 401) {
        echo "<p class='error'><strong>PROBLÈME IDENTIFIÉ :</strong> Clé API invalide</p>";
        echo "<p><strong>ACTION REQUISE :</strong> Vérifier la clé API dans config/config.php</p>";
    } elseif ($http_code === 429) {
        echo "<p class='error'><strong>PROBLÈME IDENTIFIÉ :</strong> Limite de quota dépassée</p>";
        echo "<p><strong>ACTION REQUISE :</strong> Vérifier le crédit sur platform.openai.com</p>";
    }
    exit;
}

// TEST 3 : Test avec function calling (comme NOIA)
echo "<h2>TEST 3 : Appel OpenAI avec Function Calling (comme NOIA)</h2>";
echo "<p>Question test : 'Quel est le quorum pour une reconvocation ?'</p>";

// Charger les legal facts
$db_config = [
    'host' => DB_HOST,
    'dbname' => DB_NAME,
    'user' => DB_USER,
    'password' => DB_PASSWORD
];

$legal_facts = LegalFactsManager::getRelevantFacts(
    $db_config,
    ['quorum'],
    'quorum reconvocation conseil municipal'
);

$legal_facts_section = LegalFactsManager::formatForPrompt($legal_facts);

// Pré-analyse
$pre_analysis = PreAnalysis::analyze('Quel est le quorum pour une reconvocation ?');

echo "<p>Contextes détectés : " . count($pre_analysis['detected_contexts']) . "</p>";
echo "<p>Force legal_facts : " . ($pre_analysis['force_legal_facts'] ? 'OUI' : 'NON') . "</p>";
echo "<p>Force official sources : " . ($pre_analysis['force_official_sources'] ? 'OUI' : 'NON') . "</p>";

$tools = [
    [
        'type' => 'function',
        'function' => [
            'name' => 'search_official_websites',
            'description' => 'Recherche des informations sur les sites officiels français',
            'parameters' => [
                'type' => 'object',
                'properties' => [
                    'query' => [
                        'type' => 'string',
                        'description' => 'La requête de recherche'
                    ]
                ],
                'required' => ['query']
            ]
        ]
    ]
];

$data = [
    'model' => OPENAI_MODEL,
    'messages' => [
        [
            'role' => 'system',
            'content' => $legal_facts_section . "\n\nTu es NOIA. Réponds de manière structurée avec les références légales."
        ],
        [
            'role' => 'user',
            'content' => 'Quel est le quorum pour une reconvocation ?'
        ]
    ],
    'tools' => $tools,
    'tool_choice' => 'auto',  // IMPORTANT : auto au lieu de forcer
    'max_tokens' => 1000,
    'temperature' => 0.3
];

$ch = curl_init('https://api.openai.com/v1/chat/completions');
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
curl_setopt($ch, CURLOPT_POST, true);
curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($data));
curl_setopt($ch, CURLOPT_HTTPHEADER, [
    'Content-Type: application/json',
    'Authorization: Bearer ' . OPENAI_API_KEY
]);

$start_time = microtime(true);
$response = curl_exec($ch);
$http_code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
$response_time = round((microtime(true) - $start_time) * 1000);
curl_close($ch);

echo "<p>Code HTTP : <strong>{$http_code}</strong></p>";
echo "<p>Temps de réponse : <strong>{$response_time} ms</strong></p>";

if ($http_code === 200) {
    echo "<p class='ok'>✅ Appel OpenAI avec tools réussi</p>";
    $response_data = json_decode($response, true);

    // Vérifier la structure de la réponse
    if (isset($response_data['choices'][0]['message'])) {
        $message = $response_data['choices'][0]['message'];

        echo "<h3>Structure de la réponse :</h3>";
        echo "<ul>";
        echo "<li>role : " . ($message['role'] ?? 'N/A') . "</li>";
        echo "<li>content : " . (isset($message['content']) ? '✅ PRÉSENT' : '❌ ABSENT') . "</li>";
        echo "<li>tool_calls : " . (isset($message['tool_calls']) ? '✅ PRÉSENT (' . count($message['tool_calls']) . ' appels)' : '❌ ABSENT') . "</li>";
        echo "<li>finish_reason : " . ($response_data['choices'][0]['finish_reason'] ?? 'N/A') . "</li>";
        echo "</ul>";

        if (isset($message['content'])) {
            echo "<p class='ok'><strong>CONTENU :</strong></p>";
            echo "<pre>" . htmlspecialchars($message['content']) . "</pre>";
        } else {
            echo "<p class='warning'><strong>⚠️ PAS DE CONTENU (seulement tool_calls)</strong></p>";
            echo "<p>Cela arrive quand GPT veut appeler une fonction avant de répondre.</p>";

            if (isset($message['tool_calls'])) {
                echo "<p>Tool calls demandés :</p>";
                echo "<pre>" . json_encode($message['tool_calls'], JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE) . "</pre>";
            }
        }
    }

    echo "<details><summary>Réponse complète JSON</summary><pre>" . json_encode($response_data, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE) . "</pre></details>";
} else {
    echo "<p class='error'>❌ Erreur OpenAI</p>";
    echo "<pre>" . htmlspecialchars($response) . "</pre>";
}

// TEST 4 : Vérifier le modèle utilisé
echo "<h2>TEST 4 : Configuration NOIA</h2>";
echo "<ul>";
echo "<li>Modèle : " . OPENAI_MODEL . "</li>";
echo "<li>Max tokens : " . OPENAI_MAX_TOKENS . "</li>";
echo "<li>Température : " . OPENAI_TEMPERATURE . "</li>";
echo "<li>Web search : " . (ENABLE_WEB_SEARCH ? 'activé' : 'désactivé') . "</li>";
echo "<li>RAG : " . (ENABLE_RAG ? 'activé' : 'désactivé') . "</li>";
echo "</ul>";

echo "<h2>🎯 Conclusion</h2>";
echo "<p>Si TEST 2 = ✅ mais TEST 3 montre 'PAS DE CONTENU' :</p>";
echo "<p class='warning'>➡️ <strong>PROBLÈME IDENTIFIÉ :</strong> OpenAI retourne seulement tool_calls sans content</p>";
echo "<p><strong>SOLUTION :</strong> Modifier proxy.php pour gérer les tool_calls correctement</p>";

echo "<hr>";
echo "<p><strong>⚠️ IMPORTANT :</strong> Supprimez ce fichier après diagnostic pour raisons de sécurité.</p>";
